% ================= Screenplay element macros =================

% Scene heading — bold caps, blank line before, small gap after
\newcommand{\sceneheading}[1]{%
  \needspace{4\baselineskip}%
  \par\vspace{1.2\baselineskip}%
  {\bfseries #1}\par
  \vspace{0.6\baselineskip}%
}

% Action / description — full measure, ragged right
\newcommand{\action}[1]{%
  \par\RaggedRight #1\par
  \vspace{0.6\baselineskip}%
}

% Character cue — centered over dialogue column
\newcommand{\charcue}[1]{%
  \needspace{3\baselineskip}%
  \par\hspace*{2.2in}{#1}\par
}

% Parenthetical — narrower column, indented under cue
\newcommand{\pparen}[1]{%
  \par\hspace*{1.7in}%
  \begin{minipage}[t]{2.1in}\RaggedRight (#1)\end{minipage}\par
}

% Dialogue — standard dialogue column width (~3.5in)
\newcommand{\diag}[1]{%
  \par\hspace*{1.0in}%
  \begin{minipage}[t]{3.5in}\RaggedRight #1\end{minipage}\par
  \vspace{0.6\baselineskip}%
}

% Transition — flush right, caps
\newcommand{\trans}[1]{%
  \par\hfill{#1}\par
  \vspace{0.6\baselineskip}%
}

% Sequence title marker — used between scene files, centered, underlined-style
\newcommand{\seqtitle}[1]{%
  \needspace{5\baselineskip}%
  \par\vspace{1.5\baselineskip}%
  \begin{center}\bfseries #1\end{center}
  \vspace{0.8\baselineskip}%
}
